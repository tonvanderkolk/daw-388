# 0.9.0 Kanaalklank Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Elk kanaal krijgt een 3-band EQ met concentrische knoppen, de MIC-bron krijgt de verzadiging van de 388-voorversterker en een OL-lampje meldt pieken na de EQ.

**Architecture:** De DSP zit in de engine-crate: een pure functie `preamp` in `mixer.rs` en een nieuwe module `eq.rs` met drie RBJ-piekfilters die zelf hun coëfficiënten berekenen uit frequentie en gain op de samplerate van de engine. `Channel::process` wordt bron → TRIM → (MIC: preamp) → EQ → DIRECT-aftakking → fader → PAN. Het OL-lampje leidt de UI af uit de bestaande PRE-piek (die na 0.9.0 na de EQ ligt); er komt geen nieuw atomic bij. Het projectformaat blijft versie 2: `eq` is een nieuw veld met serde-standaard (vlak).

**Tech Stack:** Rust 2024, egui/eframe 0.37, serde. Spec: `docs/superpowers/specs/2026-10-28-fase4-mixer-design.md` (Kanaal, stap 1 en 4 en 5; Testen, Klank).

```rust
/// Where the MIC preamp starts to saturate (-6 dBFS) and the level it approaches.
const PREAMP_THRESHOLD: f32 = 0.5;
const PREAMP_CEILING: f32 = 0.95;
/// Shifts the curve so the positive side saturates first: a little even-harmonic colour.
const PREAMP_BIAS: f32 = 0.05;

/// The 388's microphone preamp after TRIM: clean up to the threshold, then a soft tanh knee
/// towards the ceiling. Bias in, bias out, so the clean region passes `x` exactly and the
/// output stays below full scale.
pub(crate) fn preamp(x: f32) -> f32 {
    fn curve(x: f32) -> f32 {
        let size = x.abs();
        if size <= PREAMP_THRESHOLD {
            return x;
        }
        let room = PREAMP_CEILING - PREAMP_THRESHOLD;
        (PREAMP_THRESHOLD + room * ((size - PREAMP_THRESHOLD) / room).tanh()).copysign(x)
    }
    curve(x + PREAMP_BIAS) - curve(PREAMP_BIAS)
}
```

```rust
//! out at its own sample rate.

use crate::smooth;
use std::f64::consts::PI;

/// One band's settings.
#[derive(Debug, Clone, Copy, PartialEq)]

    pub freq: f32,
    pub gain_db: f32,

/// Flat, at the 388's centre frequencies: LOW, MID, HIGH.
pub const EQ_DEFAULTS: [EqBand; 3] = [
    EqBand { freq: 100.0, gain_db: 0.0 },
    EqBand { freq: 1000.0, gain_db: 0.0 },
    EqBand { freq: 8000.0, gain_db: 0.0 },
];

/// The bands' Q: broad and musical, like the 388's fixed-Q bands.
const Q: f64 = 0.9;
/// Coefficients are recomputed from the smoothed settings every this many samples.
const UPDATE: u32 = 32;

#[cfg(test)]
mod tests {
    use super::*;

    const RATE: u32 = 48_000;

    fn sine(freq: f32, n: usize) -> impl Iterator<Item = f32> {
        (0..n).map(move |i| 0.25 * (2.0 * std::f32::consts::PI * freq * i as f32 / RATE as f32).sin())
    }

    /// Peak of the output over the last tenth of a second, after the settings and filter settled.
    fn gain_at(bands: [EqBand; 3], freq: f32) -> f32 {
        let mut eq = Eq::new(RATE);
        eq.target = bands;
        let samples: Vec<f32> = sine(freq, RATE as usize).map(|x| eq.process(x)).collect();
        let peak = samples[samples.len() - 4800..].iter().fold(0.0f32, |p, s| p.max(s.abs()));
        20.0 * (peak / 0.25).log10()
    }

    #[test]
    fn a_band_boosts_and_cuts_15_db_at_its_frequency() {
        let mut bands = EQ_DEFAULTS;
        bands[1].gain_db = 15.0;
        assert!((gain_at(bands, 1000.0) - 15.0).abs() < 0.5);
        bands[1].gain_db = -15.0;
        assert!((gain_at(bands, 1000.0) + 15.0).abs() < 0.5);
    }

    #[test]
    fn a_flat_eq_passes_the_signal_bit_for_bit() {
        let mut eq = Eq::new(RATE);
        for x in sine(440.0, 4800) {
            assert_eq!(eq.process(x).to_bits(), x.to_bits());
        }
    }

    #[test]
    fn sweeping_a_boosted_band_does_not_click() {
        let mut eq = Eq::new(RATE);
        eq.target[1].gain_db = 15.0;
        let mut previous = 0.0f32;
        let mut biggest = 0.0f32;
        for (i, x) in sine(1000.0, 2 * RATE as usize).enumerate() {
            // Sweep MID from 500 Hz to 5 kHz in jumps, as a fast hand on the knob would.
            if i % 480 == 0 {
                eq.target[1].freq = 500.0 + 4500.0 * (i as f32 / (2 * RATE) as f32);
            }
            let y = eq.process(x);
            assert!(y.is_finite());
            if i > 0 {
                biggest = biggest.max((y - previous).abs());
            }
            previous = y;
        }
        // A 1 kHz sine of 0.25 boosted 15 dB changes at most about 0.28 per sample at 48 kHz.
        assert!(biggest < 0.4, "{biggest}");
    }
```

```rust
/// A peaking biquad (RBJ cookbook), transposed direct form II.
#[derive(Debug, Default, Clone, Copy)]
struct Biquad {
    b: [f64; 3],
    a: [f64; 2],
    state: [f64; 2],
}

impl Biquad {
    fn set(&mut self, freq: f64, gain_db: f64, rate: f64) {
        let a = 10f64.powf(gain_db / 40.0);
        let w = 2.0 * PI * freq.clamp(10.0, 0.45 * rate) / rate;
        let alpha = w.sin() / (2.0 * Q);
        let cos = w.cos();
        let a0 = 1.0 + alpha / a;
        self.b = [(1.0 + alpha * a) / a0, -2.0 * cos / a0, (1.0 - alpha * a) / a0];
        self.a = [-2.0 * cos / a0, (1.0 - alpha / a) / a0];
    }

    fn process(&mut self, x: f64) -> f64 {
        let y = self.b[0] * x + self.state[0];
        self.state[0] = self.b[1] * x - self.a[0] * y + self.state[1];
        self.state[1] = self.b[2] * x - self.a[1] * y;
        y
    }
}

/// The three bands with their smoothed settings.
#[derive(Debug)]
pub(crate) struct Eq {
    pub(crate) target: [EqBand; 3],
    current: [EqBand; 3],
    filters: [Biquad; 3],
    rate: f64,
    countdown: u32,
}

impl Eq {
    pub(crate) fn new(sample_rate: u32) -> Self {
        Self { target: EQ_DEFAULTS, current: EQ_DEFAULTS, filters: Default::default(), rate: f64::from(sample_rate), countdown: 0 }
    }

    /// Run one sample through the bands. A band that is flat and has settled is skipped (and
    /// its memory cleared), so a flat EQ passes the signal unchanged.
    pub(crate) fn process(&mut self, x: f32) -> f32 {
        let update = self.countdown == 0;
        self.countdown = if update { UPDATE - 1 } else { self.countdown - 1 };
        let mut y = f64::from(x);
        let mut touched = false;
        for ((band, target), filter) in self.current.iter_mut().zip(self.target).zip(&mut self.filters) {
            smooth(&mut band.freq, target.freq);
            smooth(&mut band.gain_db, target.gain_db);
            if band.gain_db == 0.0 && target.gain_db == 0.0 {
                *filter = Biquad::default();
                continue;
            }
            if update || filter.b == [0.0; 3] {
                filter.set(f64::from(band.freq), f64::from(band.gain_db), self.rate);
            }
            y = filter.process(y);
            touched = true;
        }
        if touched { y as f32 } else { x }
    }
}
```

Let op: `smooth` snapt binnen 1e-6 naar het doel, dus `band.gain_db` wordt exact 0.0 als het doel 0.0 is. Een overgeslagen band wordt helemaal leeggemaakt (`Biquad::default()`), dus een band die weer aangaat heeft `b == [0.0; 3]` en krijgt op zijn eerste sample meteen verse coëfficiënten, met een gain vlak bij 0 dB en een leeg geheugen: geen tik.

```rust
/// The sweep ranges of LOW, MID and HIGH, as on the 388.
const EQ_RANGES: [std::ops::RangeInclusive<f32>; 3] = [50.0..=1000.0, 500.0..=5000.0, 2500.0..=15000.0];
const EQ_NAMES: [&str; 3] = ["LOW", "MID", "HIGH"];
/// How long OL stays lit after a peak, in seconds.
const OL_HOLD: f64 = 0.5;

/// Where `freq` sits on a frequency knob's travel: logarithmic, 0 to 1.
fn freq_fraction(freq: f32, range: &std::ops::RangeInclusive<f32>) -> f32 {
    (freq / range.start()).ln() / (range.end() / range.start()).ln()
}

fn fraction_freq(fraction: f32, range: &std::ops::RangeInclusive<f32>) -> f32 {
    range.start() * (range.end() / range.start()).powf(fraction.clamp(0.0, 1.0))
}

/// OL: a peak after the EQ above -3 dBFS.
fn overload(peak: f32) -> bool {
    peak > 0.708
}

fn bands(eq: &EqEntry) -> [&BandEntry; 3] {
    [&eq.low, &eq.mid, &eq.high]
}
```

```rust
/// Which part of a concentric knob a drag started on.
#[derive(Clone, Copy, PartialEq)]
enum EqPart {
    Gain,
    Freq,
}

/// A concentric EQ knob as on the 388: the inner knob is gain (±15 dB), the outer ring the
/// frequency. A drag works on the part it started on; double-click sets that part back
/// (gain to 0 dB, frequency to its default).
fn eq_knob(ui: &mut Ui, band: &mut BandEntry, range: &std::ops::RangeInclusive<f32>, default_freq: f32, size: f32) -> egui::Response {
    let (rect, mut response) = ui.allocate_exact_size(vec2(size, size), Sense::click_and_drag());
    let centre = rect.center();
    let outer = size / 2.0 - 1.0;
    let inner = outer * 0.62;
    let part_at = |pos: egui::Pos2| if pos.distance(centre) <= inner { EqPart::Gain } else { EqPart::Freq };
    let id = response.id;
    if response.drag_started()
        && let Some(pos) = response.interact_pointer_pos()
    {
        ui.data_mut(|d| d.insert_temp(id, part_at(pos)));
    }
    if response.dragged() {
        let part = ui.data(|d| d.get_temp::<EqPart>(id)).unwrap_or(EqPart::Gain);
        let fine = if ui.input(|i| i.modifiers.shift) { 0.2 } else { 1.0 };
        let delta = -response.drag_delta().y / 150.0 * fine;
        match part {
            EqPart::Gain => band.gain_db = (band.gain_db + delta * 30.0).clamp(-15.0, 15.0),
            EqPart::Freq => band.freq_hz = fraction_freq(freq_fraction(band.freq_hz, range) + delta, range),
        }
        response.mark_changed();
    }
    if response.double_clicked()
        && let Some(pos) = response.interact_pointer_pos()
    {
        match part_at(pos) {
            EqPart::Gain => band.gain_db = 0.0,
            EqPart::Freq => band.freq_hz = default_freq,
        }
        response.mark_changed();
    }
    let painter = ui.painter();
    let pointer = |fraction: f32| {
        let angle = (-135.0 + 270.0 * fraction).to_radians();
        vec2(angle.sin(), -angle.cos())
    };
    // The ring: grey, with a light tick for the frequency.
    painter.circle_filled(centre, outer + 1.0, Color32::from_rgb(0x1b, 0x1b, 0x1c));
    painter.circle_filled(centre, outer, Color32::from_rgb(0x6e, 0x6a, 0x62));
    let f = pointer(freq_fraction(band.freq_hz, range));
    painter.line_segment([centre + f * (inner + 1.0), centre + f * outer], Stroke::new(2.0, palette::CREAM));
    // The gain knob: orange, with a dark pointer.
    painter.circle_filled(centre, inner + 1.0, Color32::from_rgb(0x1b, 0x1b, 0x1c));
    painter.circle_filled(centre, inner, palette::ORANGE);
    let g = pointer((band.gain_db + 15.0) / 30.0);
    painter.line_segment([centre + g * (inner * 0.25), centre + g * (inner * 0.9)], Stroke::new(2.0, Color32::from_rgb(0x2a, 0x26, 0x22)));
    response
}

fn format_freq(hz: f32) -> String {
    if hz >= 1000.0 { format!("{:.1} kHz", hz / 1000.0).replace('.', ",") } else { format!("{hz:.0} Hz") }
}
```
