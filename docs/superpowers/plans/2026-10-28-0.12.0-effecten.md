```rust
//! Building blocks: a fixed delay line and a Schroeder allpass.

/// Values this small in a feedback path are set to 0, so the CPU never meets denormals.
pub(crate) fn flush(x: f32) -> f32 {
    if x.abs() < 1e-15 { 0.0 } else { x }
}

/// A ring buffer of fixed length, allocated once.
#[derive(Debug, Clone)]
pub(crate) struct DelayLine {
    buffer: Vec<f32>,
    write: usize,
}

impl DelayLine {
    pub(crate) fn new(length: usize) -> Self {
        Self { buffer: vec![0.0; length.max(1)], write: 0 }
    }

    pub(crate) fn len(&self) -> usize {
        self.buffer.len()
    }

    /// The sample written `delay` samples ago (0: the last one), clamped to the length.
    pub(crate) fn tap(&self, delay: usize) -> f32 {
        let len = self.buffer.len();
        let delay = delay.min(len - 1);
        self.buffer[(self.write + len - 1 - delay) % len]
    }

    /// Linear interpolation between two taps, for a delay in fractional samples.
    pub(crate) fn tap_frac(&self, delay: f32) -> f32 {
        let whole = delay.max(0.0).floor();
        let frac = delay - whole;
        let a = self.tap(whole as usize);
        let b = self.tap(whole as usize + 1);
        a + (b - a) * frac
    }

    pub(crate) fn push(&mut self, x: f32) {
        self.buffer[self.write] = x;
        self.write = (self.write + 1) % self.buffer.len();
    }

    /// The oldest sample, then `x` in its place: a delay of the full length.
    pub(crate) fn shift(&mut self, x: f32) -> f32 {
        let out = self.buffer[self.write];
        self.push(x);
        out
    }
}

/// A Schroeder allpass of a fixed delay: flat in level, smearing in time.
#[derive(Debug, Clone)]
pub(crate) struct Allpass {
    line: DelayLine,
    pub(crate) gain: f32,
}

impl Allpass {
    pub(crate) fn new(length: usize, gain: f32) -> Self {
        Self { line: DelayLine::new(length), gain }
    }

    pub(crate) fn line(&self) -> &DelayLine {
        &self.line
    }

    pub(crate) fn process(&mut self, x: f32) -> f32 {
        let delayed = self.line.tap(self.line.len() - 1);
        let v = flush(x - self.gain * delayed);
        self.line.push(v);
        delayed + self.gain * v
    }
}
```

Tests: `DelayLine::shift` geeft een impuls na precies `len` samples terug; `tap_frac(1.5)` ligt halverwege twee samples; een allpass laat de energie van een impuls gelijk (som van kwadraten over 10 000 samples binnen 1%).

```rust
//! A plate reverb after Dattorro's topology ("Effect Design, Part 1", 1997): pre-delay,
//! input diffusion and a figure-eight tank whose taps give a decorrelated stereo output.

use crate::delay::{flush, Allpass, DelayLine};

/// The plate's controls.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct PlateParams {
    /// Reverb time in seconds, 0.5 to 6.
    pub decay_s: f32,
    /// 0 to 100 ms.
    pub predelay_ms: f32,
    /// 0 dark to 1 bright.
    pub tone: f32,
}

impl Default for PlateParams {
    fn default() -> Self {
        Self { decay_s: 2.0, predelay_ms: 20.0, tone: 0.5 }
    }
}

/// Dattorro's lengths are at 29 761 Hz.
const REFERENCE_RATE: f32 = 29_761.0;
/// Seconds one trip round half the tank takes, for turning a reverb time into a gain.
const HALF_LOOP_S: f32 = (672.0 + 4453.0 + 1800.0 + 3720.0) / REFERENCE_RATE;

#[derive(Debug, Clone)]
struct Half {
    diffuse: Allpass,
    first: DelayLine,
    damp: f32,
    second_allpass: Allpass,
    second: DelayLine,
}

/// A plate reverb: mono in, stereo out.
#[derive(Debug, Clone)]
pub struct Plate {
    pub params: PlateParams,
    rate: f32,
    predelay: DelayLine,
    bandwidth: f32,
    inputs: [Allpass; 4],
    halves: [Half; 2],
    /// The end of each half, fed into the other on the next sample.
    ends: [f32; 2],
}

impl Plate {
    pub fn new(sample_rate: u32) -> Self {
        let rate = sample_rate as f32;
        let n = |samples: f32| ((samples * rate / REFERENCE_RATE).round() as usize).max(1);
        let half = |a: f32, d1: f32, b: f32, d2: f32| Half {
            diffuse: Allpass::new(n(a), -0.7),
            first: DelayLine::new(n(d1)),
            damp: 0.0,
            second_allpass: Allpass::new(n(b), 0.5),
            second: DelayLine::new(n(d2)),
        };
        Self {
            params: PlateParams::default(),
            rate,
            predelay: DelayLine::new((0.1 * rate) as usize + 2),
            bandwidth: 0.0,
            inputs: [Allpass::new(n(142.0), 0.75), Allpass::new(n(107.0), 0.75), Allpass::new(n(379.0), 0.625), Allpass::new(n(277.0), 0.625)],
            halves: [half(672.0, 4453.0, 1800.0, 3720.0), half(908.0, 4217.0, 2656.0, 3163.0)],
            ends: [0.0; 2],
        }
    }

    pub fn process(&mut self, x: f32) -> [f32; 2] {
        let p = self.params;
        let decay = 10f32.powf(-3.0 * HALF_LOOP_S / p.decay_s.clamp(0.5, 6.0)).min(0.97);
        let damping = 0.7 - 0.65 * p.tone.clamp(0.0, 1.0);
        let wait = (p.predelay_ms.clamp(0.0, 100.0) * 0.001 * self.rate) as usize;
        self.predelay.push(x);
        let delayed = self.predelay.tap(wait);
        self.bandwidth = flush(self.bandwidth + 0.9995 * (delayed - self.bandwidth));
        let mut diffused = self.bandwidth;
        for allpass in &mut self.inputs {
            diffused = allpass.process(diffused);
        }
        let ends = self.ends;
        for (side, half) in self.halves.iter_mut().enumerate() {
            let fed = diffused + ends[1 - side] * decay;
            let a = half.diffuse.process(fed);
            let b = half.first.shift(a);
            half.damp = flush(b + damping * (half.damp - b));
            let c = half.second_allpass.process(half.damp * decay);
            self.ends[side] = half.second.shift(c);
        }
        let n = |samples: f32| (samples * self.rate / REFERENCE_RATE) as usize;
        let [l, r] = &self.halves;
        let left = r.first.tap(n(266.0)) + r.first.tap(n(2974.0)) - r.second_allpass.line().tap(n(1913.0)) + r.second.tap(n(1996.0))
            - l.first.tap(n(1990.0))
            - l.second_allpass.line().tap(n(187.0))
            - l.second.tap(n(1066.0));
        let right = l.first.tap(n(353.0)) + l.first.tap(n(3627.0)) - l.second_allpass.line().tap(n(1228.0)) + l.second.tap(n(2673.0))
            - r.first.tap(n(2111.0))
            - r.second_allpass.line().tap(n(335.0))
            - r.second.tap(n(121.0));
        [0.6 * left, 0.6 * right]
    }
}
```

Opmerking: de allpass-gain van de eerste tank-allpass is negatief (Dattorro: "decay diffusion 1" 0,7 met omgekeerd teken). Het lezen van taps uit een `DelayLine` gebruikt `tap(d)` met d vanaf het laatst geschreven sample; Dattorro meet vanaf de ingang van de lijn, dus de taps zijn een sample verschoven, wat hoorbaar niets uitmaakt. Klinkt de plate na het bouwen metalig of te kort, stem dan alleen `decay`, `damping` en de uitgangsfactor bij; de topologie blijft.

```rust
//! A tape echo: a delay with feedback through a darkening, slightly saturating loop and a
//! little wow. Turning TIME glides like a changing tape speed.

use crate::delay::{flush, DelayLine};

/// TIME follows the metronome when not `Off`.
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub enum EchoSync {
    #[default]
    Off,
    Quarter,
    Eighth,
    DottedEighth,
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub struct EchoParams {
    /// 40 to 800 ms when SYNC is off.
    pub time_ms: f32,
    pub sync: EchoSync,
    /// 0 to 0.9.
    pub feedback: f32,
    /// 0 dark to 1 bright.
    pub tone: f32,
}

impl Default for EchoParams {
    fn default() -> Self {
        Self { time_ms: 300.0, sync: EchoSync::Off, feedback: 0.35, tone: 0.5 }
    }
}

const MAX_SECONDS: f32 = 2.0;

impl EchoParams {
    /// The delay in seconds; `quarter` is a quarter note in seconds at the current tempo.
    pub fn seconds(&self, quarter: f32) -> f32 {
        let seconds = match self.sync {
            EchoSync::Off => self.time_ms.clamp(40.0, 800.0) / 1000.0,
            EchoSync::Quarter => quarter,
            EchoSync::Eighth => quarter / 2.0,
            EchoSync::DottedEighth => quarter * 0.75,
        };
        seconds.clamp(0.001, MAX_SECONDS)
    }
}

#[derive(Debug, Clone)]
pub struct Echo {
    pub params: EchoParams,
    rate: f32,
    line: DelayLine,
    /// The delay in samples, gliding towards the target.
    delay: f32,
    lowpass: f32,
    highpass: [f32; 2],
    phase: f32,
}

impl Echo {
    pub fn new(sample_rate: u32) -> Self {
        let rate = sample_rate as f32;
        Self { params: EchoParams::default(), rate, line: DelayLine::new((MAX_SECONDS * rate) as usize + 64), delay: 0.0, lowpass: 0.0, highpass: [0.0; 2], phase: 0.0 }
    }

    /// Jump to the target delay at once (a new project, tests).
    pub fn settle(&mut self) {
        self.delay = self.params.seconds(0.5) * self.rate;
    }

    /// One sample; `quarter` is a quarter note in seconds at the current tempo.
    pub fn process(&mut self, x: f32, quarter: f32) -> f32 {
        let target = self.params.seconds(quarter) * self.rate;
        // About a third of a second to glide: a tape motor finding its new speed.
        self.delay += (target - self.delay) * (1.0 / (0.3 * self.rate));
        self.phase = (self.phase + 0.4 / self.rate).fract();
        let wow = 0.0003 * self.rate * (std::f32::consts::TAU * self.phase).sin();
        let delayed = self.line.tap_frac((self.delay + wow).max(1.0) - 1.0);
        // The loop: lowpass (TONE), a fixed highpass at 100 Hz, soft saturation.
        let cutoff = 1_500.0 + 7_500.0 * self.params.tone.clamp(0.0, 1.0);
        let a = 1.0 - (-std::f32::consts::TAU * cutoff / self.rate).exp();
        self.lowpass = flush(self.lowpass + a * (delayed - self.lowpass));
        let h = (-std::f32::consts::TAU * 100.0 / self.rate).exp();
        let high = h * (self.highpass[1] + self.lowpass - self.highpass[0]);
        self.highpass = [self.lowpass, flush(high)];
        let fed = (high * self.params.feedback.clamp(0.0, 0.9)).tanh();
        self.line.push(flush(x + fed));
        delayed
    }
}
```
