//! Energy meters: the audio thread adds up squares, the UI reads the mean and starts over.

use std::sync::atomic::{AtomicU32, AtomicU64, Ordering};

/// The energy of a signal since the UI last read it: the sum of squares (as `f64` bits) and the
/// number of samples. Sum and count are two atomics, so a read can split one block between two
/// readings; the meter's own slowness hides that.
#[derive(Debug, Default)]
pub(crate) struct Energy {
    sum: AtomicU64,
    count: AtomicU32,
}

impl Energy {
    /// Add `sum` (a sum of squares) over `count` samples.
    pub(crate) fn add(&self, sum: f64, count: u32) {
        let mut current = self.sum.load(Ordering::Relaxed);
        loop {
            let next = (f64::from_bits(current) + sum).to_bits();
            match self.sum.compare_exchange_weak(current, next, Ordering::Relaxed, Ordering::Relaxed) {
                Ok(_) => break,
                Err(actual) => current = actual,
            }
        }
        self.count.fetch_add(count, Ordering::Relaxed);
    }

    /// The mean square since the last call (0 without samples), and start over.
    pub(crate) fn take(&self) -> f32 {
        let count = self.count.swap(0, Ordering::Relaxed);
        let sum = f64::from_bits(self.sum.swap(0, Ordering::Relaxed));
        if count == 0 { 0.0 } else { (sum / f64::from(count)) as f32 }
    }
}

#[cfg(test)]

    #[test]
    fn takes_the_mean_square_and_starts_over() {
        let energy = Energy::default();
        assert_eq!(energy.take(), 0.0, "no samples");
        energy.add(2.0, 4);
        energy.add(1.0, 4);
        assert!((energy.take() - 0.375).abs() < 1e-6);

//! A VU meter: the needle's ballistics, the mapping from energy to needle position and the drawing.

#![cfg_attr(not(test), allow(dead_code))] // Used from Task 6 on.

/// 0 VU in dBFS RMS, the usual calibration for 24-bit recording.
pub const REFERENCE_DBFS: f32 = -18.0;
/// The scale, in VU.
pub const SCALE_MIN: f32 = -20.0;
pub const SCALE_MAX: f32 = 3.0;
/// Damping ratio: a little under critical, for the VU's slight overshoot (about 1%).
const DAMPING: f32 = 0.81;
/// Natural frequency in rad/s: within 1% of a step after 300 ms.
const OMEGA: f32 = 22.0;
/// Integration step in seconds, small enough to stay stable at any frame rate.
const SUBSTEP: f32 = 0.001;

/// Needle position (0 at -20 VU, 1 at +3 VU) for a level in VU. A real VU's deflection follows
/// the voltage, so the scale is linear in amplitude, not in dB.
pub fn position(vu: f32) -> f32 {
    let amplitude = |vu: f32| 10f32.powf(vu / 20.0);
    let (low, high) = (amplitude(SCALE_MIN), amplitude(SCALE_MAX));
    (amplitude(vu.clamp(SCALE_MIN, SCALE_MAX)) - low) / (high - low)
}

/// Needle position for a mean square from the engine; silence rests at -20.
pub fn target(mean_square: f32) -> f32 {
    if mean_square <= 0.0 {
        return 0.0;
    }
    position(10.0 * mean_square.log10() - REFERENCE_DBFS)
}

/// A VU needle: a damped spring towards its target, against the stops at both ends.
#[derive(Debug, Default, Clone, Copy)]
pub struct Needle {
    position: f32,
    velocity: f32,
}

impl Needle {
    /// Move the needle `dt` seconds towards `target` (a position from [`target`]).
    pub fn step(&mut self, dt: f32, target: f32) {
        let steps = (dt / SUBSTEP).ceil().max(1.0) as usize;
        let h = dt / steps as f32;
        for _ in 0..steps {
            let accel = OMEGA * OMEGA * (target - self.position) - 2.0 * DAMPING * OMEGA * self.velocity;
            self.velocity += accel * h;
            self.position += self.velocity * h;
            if self.position < 0.0 {
                self.position = 0.0;
                self.velocity = self.velocity.max(0.0);
            } else if self.position > 1.0 {
                self.position = 1.0;
                self.velocity = self.velocity.min(0.0);
            }
        }
    }

    /// Where the needle stands, 0 to 1.
    pub fn position(&self) -> f32 {
        self.position
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    /// Run a needle at 60 frames per second.
    fn run(needle: &mut Needle, seconds: f32, target: f32) -> f32 {
        let mut highest = needle.position();
        for _ in 0..(seconds * 60.0).round() as usize {
            needle.step(1.0 / 60.0, target);
            highest = highest.max(needle.position());
        }
        highest
    }

    #[test]
    fn minus_18_dbfs_rms_is_0_vu() {
        // A sine with a peak of -15 dBFS has an RMS of -18 dBFS.
        let mean_square = 10f32.powf(-18.0 / 10.0);
        assert!((target(mean_square) - position(0.0)).abs() < 1e-4);
        assert!(position(0.0) > 0.6 && position(0.0) < 0.75, "0 VU at about two thirds");
    }

    #[test]
    fn reaches_a_step_in_300_ms_with_a_little_overshoot() {
        let goal = position(0.0);
        let mut needle = Needle::default();
        let highest = run(&mut needle, 0.3, goal);
        assert!((needle.position() - goal).abs() < 0.01 * goal, "{} after 300 ms", needle.position());
        let highest = highest.max(run(&mut needle, 1.0, goal));
        assert!(highest > goal, "a VU overshoots a little");
        assert!(highest <= goal * 1.015, "overshoot {highest}");
    }

    #[test]
    fn falls_back_to_minus_20_in_silence() {
        let mut needle = Needle::default();
        run(&mut needle, 1.0, position(0.0));
        run(&mut needle, 1.0, target(0.0));
        assert_eq!(needle.position(), 0.0);
    }

    #[test]
    fn stays_on_the_stop_above_plus_3() {
        let mut needle = Needle::default();
        let highest = run(&mut needle, 2.0, target(1.0));
        assert_eq!(needle.position(), 1.0);
        assert!(highest <= 1.0);
    }
}

//! The tape deck: the two reels and the line about the tape.

#![cfg_attr(not(test), allow(dead_code))] // Used from Task 6 on.

use daw388_project::session::{KindEntry, SpeedEntry, TapeMode, TapeSettings};
use std::f32::consts::TAU;

/// The tape runs past at 7½ ips, as on the 388.
const IPS: f32 = 7.5;
/// Hub and full-pack radius of a 7-inch reel, in inches.
pub const CORE: f32 = 1.0;
pub const FULL: f32 = 3.4;
/// Fastest drawn rotation, in turns per second, so winding does not strobe.
const MAX_TURNS: f32 = 3.0;
/// A jump in tape position larger than this (RTZ, a click in the drawer) is not winding.
const JUMP: f64 = 2.0;
/// The shortest tape the reels show, in seconds.
pub const MIN_LENGTH: f64 = 180.0;

/// The reels' angles (radians, clockwise) and how far the tape has run.
#[derive(Debug, Default, Clone)]
pub struct Reels {
    pub left_angle: f32,
    pub right_angle: f32,
    /// Fraction of the tape on the right reel, 0 to 1.
    pub wound: f32,
    last_position: Option<f64>,
}

impl Reels {
    /// Follow the tape to `position` (seconds from the start) on a tape of `length` seconds,
    /// `dt` seconds after the previous call.
    pub fn advance(&mut self, position: f64, length: f64, dt: f32) {
        let length = length.max(MIN_LENGTH);
        self.wound = (position / length).clamp(0.0, 1.0) as f32;
        let moved = self.last_position.map_or(0.0, |last| position - last);
        self.last_position = Some(position);
        if moved.abs() > JUMP {
            return;
        }
        let inches = moved as f32 * IPS;
        let limit = MAX_TURNS * TAU * dt;
        let (left, right) = self.radii();
        self.left_angle = (self.left_angle + (inches / left).clamp(-limit, limit)) % TAU;
        self.right_angle = (self.right_angle + (inches / right).clamp(-limit, limit)) % TAU;
    }

    /// Radius of the tape pack on the left and the right reel, in inches. The tape's area is
    /// what moves from one reel to the other.
    pub fn radii(&self) -> (f32, f32) {
        let area = FULL * FULL - CORE * CORE;
        let radius = |share: f32| (CORE * CORE + share * area).sqrt();
        (radius(1.0 - self.wound), radius(self.wound))
    }
}

/// The line above the reels: "456-stijl · 7½ IPS · dbx aan", or "Clean".
pub fn tape_line(tape: &TapeSettings) -> String {
    if tape.mode == TapeMode::Clean {
        return "Clean".into();
    }
    let kind = match tape.kind {
        KindEntry::T456 => "456",
        KindEntry::T250 => "250",
        KindEntry::T911 => "911",
    };
    let speed = match tape.speed {
        SpeedEntry::Ips3_75 => "3¾",
        SpeedEntry::Ips7_5 => "7½",
        SpeedEntry::Ips15 => "15",
    };
    let dbx = match tape.dbx {
        [true, true] => "dbx aan",
        [true, false] => "dbx 1-4",
        [false, true] => "dbx 5-8",
        [false, false] => "dbx uit",
    };
    format!("{kind}-stijl · {speed} IPS · {dbx}")
}

//! IBM Plex Sans Condensed and IBM Plex Mono (SIL Open Font License, see `assets/fonts/OFL.txt`).

#![cfg_attr(not(test), allow(dead_code))] // Used from Task 6 on.

use egui::{FontData, FontDefinitions, FontFamily, RichText};
use std::sync::Arc;

const SANS: &str = "plex-sans";
const BOLD: &str = "plex-bold";
const MONO: &str = "plex-mono";

/// egui's fonts with Plex in front: Proportional is Plex Sans Condensed, Monospace Plex Mono
/// (the LED counter) and the family [`BOLD`] Plex Sans Condensed SemiBold. egui's own fonts stay
/// behind them for characters Plex lacks.
pub fn definitions() -> FontDefinitions {
    let mut fonts = FontDefinitions::default();
    for (name, bytes) in [
        (SANS, &include_bytes!("../../assets/fonts/IBMPlexSansCondensed-Regular.ttf")[..]),
        (BOLD, &include_bytes!("../../assets/fonts/IBMPlexSansCondensed-SemiBold.ttf")[..]),
        (MONO, &include_bytes!("../../assets/fonts/IBMPlexMono-Medium.ttf")[..]),
    ] {
        fonts.font_data.insert(name.into(), Arc::new(FontData::from_static(bytes)));
    }
    let fallback = fonts.families.get(&FontFamily::Proportional).cloned().unwrap_or_default();
    let family = |front: &str, base: Vec<String>| [vec![front.to_string()], base].concat();
    let mono_base = fonts.families.get(&FontFamily::Monospace).cloned().unwrap_or_default();
    fonts.families.insert(FontFamily::Proportional, family(SANS, fallback.clone()));
    fonts.families.insert(FontFamily::Name(BOLD.into()), family(BOLD, fallback));
    fonts.families.insert(FontFamily::Monospace, family(MONO, mono_base));
    fonts
}

/// Install the fonts.
pub fn install(ctx: &egui::Context) {
    ctx.set_fonts(definitions());
}

/// Text in Plex SemiBold: egui has no weight per text, so bold is its own family.
pub fn bold(text: impl Into<String>) -> RichText {
    RichText::new(text).family(FontFamily::Name(BOLD.into()))
}

        assert_eq!(fonts.families[&FontFamily::Monospace][0], MONO);
        assert_eq!(fonts.families[&FontFamily::Name(BOLD.into())][0], BOLD);

use crate::palette;
use egui::{Align2, Color32, FontId, Pos2, Stroke, Ui, Vec2, pos2, vec2};

/// The scale marks, in VU.
const MARKS: [f32; 11] = [-20.0, -10.0, -7.0, -5.0, -3.0, -2.0, -1.0, 0.0, 1.0, 2.0, 3.0];
/// The scale's sweep either side of straight up, in radians.
const SWEEP: f32 = 0.85;

/// Draw a VU meter of `size` with the needle at `needle` (0 to 1) and the OL lamp.
pub fn draw(ui: &mut Ui, size: Vec2, needle: f32, ol: bool) -> egui::Response {
    let (rect, response) = ui.allocate_exact_size(size, egui::Sense::hover());
    let painter = ui.painter().with_clip_rect(rect);
    painter.rect_filled(rect, 3.0, palette::EDGE);
    let face = rect.shrink(3.0);
    painter.rect_filled(face, 2.0, palette::VU_FACE);
    // The pivot sits under the face, so only the top of the arc shows, as on a real VU.
    let pivot = pos2(face.center().x, face.bottom() + face.height() * 0.35);
    let radius = face.height() * 1.05;
    let at = |fraction: f32, r: f32| -> Pos2 {
        let angle = -SWEEP + 2.0 * SWEEP * fraction;
        pivot + vec2(angle.sin(), -angle.cos()) * r
    };
    let arc = |from: f32, to: f32, r: f32, stroke: Stroke| {
        let points: Vec<Pos2> = (0..=24).map(|i| at(from + (to - from) * i as f32 / 24.0, r)).collect();
        painter.add(egui::Shape::line(points, stroke));
    };
    let zero = position(0.0);
    arc(0.0, zero, radius, Stroke::new(1.2, palette::VU_INK));
    arc(zero, 1.0, radius, Stroke::new(3.0, palette::VU_RED));
    for mark in MARKS {
        let fraction = position(mark);
        let colour = if mark > 0.0 { palette::VU_RED } else { palette::VU_INK };
        painter.line_segment([at(fraction, radius), at(fraction, radius + face.height() * 0.12)], Stroke::new(1.0, colour));
    }
    let small = FontId::proportional((face.height() * 0.16).clamp(7.0, 12.0));
    painter.text(at(0.5, radius * 0.72), Align2::CENTER_CENTER, "VU", FontId::proportional((face.height() * 0.2).clamp(8.0, 15.0)), palette::VU_INK);
    painter.text(at(0.0, radius + face.height() * 0.24), Align2::CENTER_CENTER, "-20", small.clone(), palette::VU_INK);
    painter.text(at(zero, radius + face.height() * 0.24), Align2::CENTER_CENTER, "0", small.clone(), palette::VU_RED);
    painter.text(at(1.0, radius + face.height() * 0.24), Align2::CENTER_CENTER, "+3", small, palette::VU_RED);
    painter.line_segment([pivot, at(needle, radius + face.height() * 0.1)], Stroke::new(1.4, Color32::BLACK));
    if ol {
        painter.circle_filled(pos2(face.right() - 7.0, face.top() + 7.0), 3.5, palette::LED);
    }
    painter.rect_stroke(face, 2.0, Stroke::new(1.0, palette::EDGE), egui::StrokeKind::Inside);
    response
}

use egui::{Align2, FontId, Pos2, Stroke, Ui, Vec2, pos2, vec2};

/// The size of the tape deck's window at zoom 1.
pub const WINDOW: Vec2 = Vec2::new(262.0, 190.0);

/// Draw the smoked-glass window with both reels and the line about the tape.
pub fn draw(ui: &mut Ui, reels: &Reels, line: &str) {
    let (rect, _) = ui.allocate_exact_size(WINDOW, egui::Sense::hover());
    let painter = ui.painter().with_clip_rect(rect);
    painter.rect_filled(rect, 4.0, palette::EDGE);
    let glass = rect.shrink(4.0);
    painter.rect_filled(glass, 3.0, palette::GLASS);
    painter.text(glass.left_top() + vec2(10.0, 8.0), Align2::LEFT_TOP, line, FontId::proportional(11.0), palette::TEXT_DIM);
    // Inches to points: a full 7-inch reel nearly fills half the window.
    let scale = (glass.width() / 2.0 - 12.0) / (FULL + 0.2);
    let y = glass.center().y + 10.0;
    let centres = [pos2(glass.left() + glass.width() * 0.27, y), pos2(glass.left() + glass.width() * 0.73, y)];
    let (left, right) = reels.radii();
    // The tape between the packs, along the bottom.
    let bottom = |centre: Pos2, r: f32| centre + vec2(0.0, r * scale);
    painter.line_segment([bottom(centres[0], left), bottom(centres[1], right)], Stroke::new(1.5, palette::TAPE_PACK));
    for (centre, pack, angle) in [(centres[0], left, reels.left_angle), (centres[1], right, reels.right_angle)] {
        painter.circle_filled(centre, (FULL + 0.15) * scale, palette::REEL_FLANGE.gamma_multiply(0.55));
        painter.circle_filled(centre, pack * scale, palette::TAPE_PACK);
        painter.circle_filled(centre, CORE * scale, palette::REEL);
        // Three spokes: the openings in the flange turn with the reel.
        for spoke in 0..3 {
            let a = angle + spoke as f32 * std::f32::consts::TAU / 3.0;
            let tip = centre + vec2(a.sin(), -a.cos()) * (FULL * scale * 0.95);
            painter.line_segment([centre, tip], Stroke::new(CORE * scale * 0.55, palette::REEL));
        }
        painter.circle_filled(centre, CORE * scale * 0.35, palette::REEL_FLANGE);
        painter.circle_filled(centre, CORE * scale * 0.12, palette::GLASS);
    }
}

        let Some(engine) = &mut self.engine else { return };
        engine.collect_garbage();
        // VU-like fall of about 20 dB per second for the TRK bars of the MONITOR section.
        let fall = 20.0 * dt;

            let peaks = engine.take_track_peaks(index);
            // The PRE peak lies after the EQ: where the 388's OL looks.
            if overload(peaks.pre) {
                track.ol_until = now + OL_HOLD;
            }

            track.vu.step(dt, skin::vu::target(if self.meter_post { energy.post } else { energy.pre }));

        let (master_peaks, send_peaks) = (engine.take_master_peaks(), engine.take_send_peaks());
