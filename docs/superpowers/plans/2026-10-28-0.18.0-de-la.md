```rust
//! A waveform overview: the lowest and highest sample per block, so a region draws quickly at
//! any zoom.

/// Samples per block.
pub const BLOCK: usize = 256;

#[derive(Debug, Clone, Default, PartialEq)]
pub struct Peaks {
    /// The source's sample rate.
    pub rate: u32,
    /// Lowest and highest sample of each block.
    pub blocks: Vec<(f32, f32)>,
}

impl Peaks {
    /// The overview of `samples` at `rate`.
    pub fn of(samples: &[f32], rate: u32) -> Self {
        let blocks = samples.chunks(BLOCK).map(|c| c.iter().fold((0.0f32, 0.0f32), |(lo, hi), s| (lo.min(*s), hi.max(*s)))).collect();
        Self { rate, blocks }
    }

    /// Lowest and highest sample between `from` and `to` seconds in the source; at least one
    /// block, so a very close zoom still shows something.
    pub fn range(&self, from: f64, to: f64) -> (f32, f32) {
        let per_second = f64::from(self.rate) / BLOCK as f64;
        let first = (from.max(0.0) * per_second).floor() as usize;
        if first >= self.blocks.len() {
            return (0.0, 0.0);
        }
        let last = ((to * per_second).ceil() as usize).max(first + 1).min(self.blocks.len());
        self.blocks[first..last].iter().fold((0.0, 0.0), |(lo, hi), (l, h)| (lo.min(*l), hi.max(*h)))
    }
}
```

Tests: `of` op 600 samples geeft 3 blokken; een sinus van amplitude 0,5 (periode 64 samples) geeft per blok ongeveer -0,5 en 0,5 (binnen 0,01); voor 1024 samples op 1024 Hz die in de eerste helft 0,25 en in de tweede helft 1,0 zijn, geeft `range(0.0, 0.49)` `(0.0, 0.25)`; `range` voorbij het einde geeft `(0.0, 0.0)`; `range(x, x)` geeft één blok. Geef `Peaks` en zijn velden doc-comments.

```rust
//! The drawer's time axis: seconds and pixels, the beat grid, snapping, and which region lies
//! under the pointer. Pure, so it is tested without a window.

use daw388_project::regions::Region;

/// How close (in pixels) to a region's edge the pointer grabs the edge instead of the region.
pub(crate) const EDGE_GRIP: f32 = 6.0;
/// How far (in pixels) snapping reaches.
pub(crate) const SNAP_PIXELS: f32 = 8.0;
/// Zoom limits in pixels per second.
pub(crate) const ZOOM_MIN: f64 = 2.0;
pub(crate) const ZOOM_MAX: f64 = 20_000.0;

/// Where time sits on screen: x `left` shows `start` seconds, `zoom` pixels per second.
#[derive(Debug, Clone, Copy, PartialEq)]
pub(crate) struct Timeline {
    pub left: f32,
    pub start: f64,
    pub zoom: f64,
}

impl Timeline {
    /// The x of a time.
    pub fn x(&self, seconds: f64) -> f32 {
        self.left + ((seconds - self.start) * self.zoom) as f32
    }

    /// The time at an x.
    pub fn seconds(&self, x: f32) -> f64 {
        self.start + f64::from(x - self.left) / self.zoom
    }

    /// The zoom that shows `length` seconds (at least 10) in `width` pixels.
    pub fn fit(length: f64, width: f32) -> f64 {
        (f64::from(width.max(1.0)) / length.max(10.0)).clamp(ZOOM_MIN, ZOOM_MAX)
    }

    /// The new start after zooming to `zoom` with the time under `x` staying put.
    pub fn zoomed_start(&self, zoom: f64, x: f32) -> f64 {
        (self.seconds(x) - f64::from(x - self.left) / zoom).max(0.0)
    }
}

/// A beat and a bar in seconds. The tempo counts quarter notes; in x/8 a beat is an eighth,
/// as the metronome does.
pub(crate) fn beat_and_bar(bpm: f64, beats: u8, unit: u8) -> (f64, f64) {
    let beat = 60.0 / bpm.max(1.0) * 4.0 / f64::from(unit.max(1));
    (beat, beat * f64::from(beats.max(1)))
}

/// The beats from `from` to `to` (bar 1 starts at 0:00): their time and their bar number when a
/// bar starts there.
pub(crate) fn beats(from: f64, to: f64, beat: f64, per_bar: u8) -> Vec<(f64, Option<u32>)> {
    let per_bar = i64::from(per_bar.max(1));
    let first = (from.max(0.0) / beat).ceil() as i64;
    let last = (to.max(0.0) / beat).floor() as i64;
    (first..=last).map(|n| (n as f64 * beat, (n % per_bar == 0).then(|| (n / per_bar + 1) as u32))).collect()
}

/// The nearest of `marks` within `reach` of `t`, else `t`.
pub(crate) fn snap(t: f64, marks: &[f64], reach: f64) -> f64 {
    marks.iter().copied().filter(|m| (m - t).abs() <= reach).min_by(|a, b| (a - t).abs().total_cmp(&(b - t).abs())).unwrap_or(t)
}

/// A shift `delta` adjusted so that the one of `edges` that is closest to a mark lands on it.
pub(crate) fn snap_shift(edges: &[f64], delta: f64, marks: &[f64], reach: f64) -> f64 {
    let mut best: Option<f64> = None;
    for edge in edges {
        let moved = edge + delta;
        let adjust = snap(moved, marks, reach) - moved;
        if adjust != 0.0 && best.is_none_or(|b| adjust.abs() < b.abs()) {
            best = Some(adjust);
        }
    }
    delta + best.unwrap_or(0.0)
}

/// Which edge of a region.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum Edge {
    Start,
    End,
}

/// What lies under the pointer in a lane.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum Hit {
    /// A region, and which edge when the pointer is on one.
    Region { index: usize, edge: Option<Edge> },
    Empty,
}

/// What lies under `x` on a track. A region narrower than three grips has no edges to grab.
pub(crate) fn hit(line: &Timeline, regions: &[Region], x: f32) -> Hit {
    for (index, region) in regions.iter().enumerate() {
        let (left, right) = (line.x(region.start), line.x(region.end()));
        if x < left || x > right {
            continue;
        }
        let wide = right - left > 3.0 * EDGE_GRIP;
        let edge = if wide && x - left <= EDGE_GRIP {
            Some(Edge::Start)
        } else if wide && right - x <= EDGE_GRIP {
            Some(Edge::End)
        } else {
            None
        };
        return Hit::Region { index, edge };
    }
    Hit::Empty
}

/// The lane under `y`, with lanes of `height` from `top`.
pub(crate) fn lane_at(top: f32, height: f32, lanes: usize, y: f32) -> Option<usize> {
    let lane = ((y - top) / height).floor();
    (lane >= 0.0 && (lane as usize) < lanes).then_some(lane as usize)
}

/// A step for the seconds ruler: at least `min_pixels` apart.
pub(crate) fn seconds_step(zoom: f64, min_pixels: f64) -> f64 {
    [0.1, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0, 15.0, 30.0, 60.0, 120.0, 300.0].into_iter().find(|s| s * zoom >= min_pixels).unwrap_or(600.0)
}
```

```rust
/// Like [`Self::push`], but a change in the same `group` as the last step joins that step
/// (files dropped together arrive one by one, and are still one step).
pub fn push_grouped(&mut self, name: impl Into<String>, group: u64, tracks: Vec<(usize, TrackContent)>) {
    if let Some(last) = self.undo.last_mut().filter(|s| s.group == Some(group)) {
        for (track, content) in tracks {
            if !last.tracks.iter().any(|(t, _)| *t == track) {
                last.tracks.push((track, content));
            }
        }
        self.redo.clear();
        return;
    }
    self.push(name, tracks);
    if let Some(last) = self.undo.last_mut() {
        last.group = Some(group);
    }
}
```

```rust
//! Edits from the drawer: each one is a single undo step built from the pure functions in
//! `regions.rs`.

use super::*;
use crate::sources::SourceState;
use crate::timeline::Edge;
use daw388_project::regions::{self, Clipboard};

impl DawApp {
    /// Whether editing is possible now; says why not.
    fn may_edit(&mut self) -> bool {
        if self.recording_now() {
            self.status = "Bewerken kan niet tijdens een opname".into();
            return false;
        }
        true
    }

    /// Whether `tracks` can take new regions (a track waiting for a file cannot); says why not.
    fn tracks_free(&mut self, tracks: impl IntoIterator<Item = usize>) -> bool {
        tracks.into_iter().all(|t| !self.refuse_unresolved(t))
    }

    fn all_regions(&self) -> Vec<Vec<Region>> {
        self.tracks.iter().map(|t| t.regions.clone()).collect()
    }

    /// Put new region lists on the tracks as one step (only tracks that changed are in it).
    fn set_regions(&mut self, name: &str, new: Vec<Vec<Region>>) {
        let changed: Vec<usize> = (0..TRACKS).filter(|&t| self.tracks[t].regions != new[t]).collect();
        if changed.is_empty() {
            return;
        }
        self.change(name, &changed, |app| {
            for &t in &changed {
                app.tracks[t].regions.clone_from(&new[t]);
            }
        });
    }

    /// Select these regions again after an edit (they are found by their values).
    fn select(&mut self, placed: &[(usize, Region)]) {
        self.editor.selection = placed
            .iter()
            .filter_map(|(t, r)| self.tracks.get(*t)?.regions.iter().position(|x| x == r).map(|i| (*t, i)))
            .collect();
    }

    fn playhead(&self) -> f64 {
        self.engine.as_ref().map_or(0.0, EngineHandle::position_seconds).max(0.0)
    }

    /// Indices of the selection per track, highest first, so removing them keeps the rest valid.
    fn selection_by_track(&self) -> Vec<(usize, usize)> {
        let mut selection = self.editor.selection.clone();
        selection.sort_unstable_by(|a, b| b.cmp(a));
        selection.dedup();
        selection
    }
}
```
