```rust
//! The MONITOR section: per track a GAIN and PAN, and the 388's switching logic that decides
//! whether a track's monitor hears its input (the PGM bus or the DIRECT tap), the tape or
//! nothing. The table is in the 388's manual under "Monitor Source with MONITOR/PHONES Selector
//! in STEREO".

use crate::mixer::pan_gains;
use crate::smooth;

/// The MON switch: CUE follows the switching logic, PGM always hears the buses.
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub enum MonitorMode {
    Pgm,
    #[default]
    Cue,
}

/// One track's MONITOR controls.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct MonitorParams {
    /// Linear gain.
    pub gain: f32,
    /// -1.0 is hard left, 1.0 is hard right.
    pub pan: f32,
    pub mute: bool,
}

impl Default for MonitorParams {
    fn default() -> Self {
        Self { gain: 1.0, pan: 0.0, mute: false }
    }
}

/// What a track's monitor hears.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum MonitorSource {
    Input,
    Tape,
    Silent,
}

/// The track hears its input rather than the tape: it is recording, or it is armed with INSERT
/// off. A RMX channel of such a track stays silent, so bus, track and channel never form a loop.
pub(crate) fn hears_input(armed: bool, insert: bool, recording: bool) -> bool {
    armed && (recording || !insert)
}

/// The 388's table: PGM always hears the input; CUE hears the input while the track hears its
/// input, else the tape while it plays, else nothing.
pub(crate) fn monitor_source(mode: MonitorMode, armed: bool, insert: bool, recording: bool, tape_playing: bool) -> MonitorSource {
    match mode {
        MonitorMode::Pgm => MonitorSource::Input,
        MonitorMode::Cue if hears_input(armed, insert, recording) => MonitorSource::Input,
        MonitorMode::Cue if tape_playing => MonitorSource::Tape,
        MonitorMode::Cue => MonitorSource::Silent,
    }
}

/// A track's monitor with its smoothed left and right gains.
#[derive(Debug, Default)]
pub(crate) struct Monitor {
    pub(crate) params: MonitorParams,
    current: [f32; 2],
}

impl Monitor {
    /// This frame's left and right gains, moved one step towards their target.
    pub(crate) fn gains(&mut self) -> [f32; 2] {
        let target = if self.params.mute { [0.0; 2] } else { pan_gains(self.params.gain, self.params.pan) };
        smooth(&mut self.current[0], target[0]);
        smooth(&mut self.current[1], target[1]);
        self.current
    }
}

mod tests {
```

```rust
//! The 388's input channels: a source (MIC, LINE or the tape in the RMX position), TRIM, the
//! channel fader, PAN and the taps the buses take from. EQ and the sends follow in 0.9 and 0.10.

use crate::monitor::MonitorMode;
use crate::smooth;
use std::f32::consts::FRAC_PI_4;

/// Where a channel's signal comes from.
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub enum Source {
    /// The channel's input through the microphone preamp (its character follows in 0.9).
    Mic,
    /// The channel's input, clean.
    Line,
    /// The playback of the track with the same number, the 388's remix position.
    #[default]
    Remix,
}

/// One channel's settings, sent whole with [`crate::Command::SetChannel`].
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct ChannelParams {
    pub source: Source,
    /// The input channel for MIC and LINE, counted from 0.
    pub input: Option<u16>,
    /// Linear TRIM gain.
    pub trim: f32,
    /// Linear fader gain.
    pub fader: f32,
    /// -1.0 is hard left (the odd buses), 1.0 hard right (the even buses).
    pub pan: f32,
    pub mute: bool,
    /// ASSIGN 1-2, 3-4, 5-6 and 7-8.
    pub assign: [bool; 4],
    /// ASSIGN L-R: to the stereo bus.
    pub lr: bool,
    /// The track with the same number records this channel after TRIM instead of its bus.
    pub direct: bool,
}

impl Default for ChannelParams {
    fn default() -> Self {
        Self { source: Source::Remix, input: None, trim: 1.0, fader: 1.0, pan: 0.0, mute: false, assign: [false; 4], lr: false, direct: false }
    }
}

impl ChannelParams {
    /// MIC or LINE with an input the device has: a live source.
    pub(crate) fn is_live(&self, input_channels: usize) -> bool {
        self.source != Source::Remix && self.input.is_some_and(|c| usize::from(c) < input_channels)
    }
}

/// The master section's settings, sent whole with [`crate::Command::SetMixer`].
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct MixerParams {
    /// Linear gains of the PGM MASTERS: ODD (buses 1, 3, 5, 7) and EVEN (2, 4, 6, 8).
    pub odd: f32,
    pub even: f32,
    pub mon: MonitorMode,
    /// Left and right averaged on both sides after the STEREO MASTER.
    pub mono: bool,
}

impl Default for MixerParams {
    fn default() -> Self {
        Self { odd: 1.0, even: 1.0, mon: MonitorMode::Cue, mono: false }
    }
}

/// Constant-power pan: -3 dB per side in the centre.
pub(crate) fn pan_gains(gain: f32, pan: f32) -> [f32; 2] {
    let angle = (pan.clamp(-1.0, 1.0) + 1.0) * FRAC_PI_4;
    [gain * angle.cos(), gain * angle.sin()]
}

/// What a channel hands on for one frame.
#[derive(Debug, Clone, Copy, Default, PartialEq)]
pub(crate) struct Taps {
    /// After TRIM, before the fader: DIRECT and the PRE meter.
    pub(crate) direct: f32,
    /// After fader and mute, before PAN: the POST meter.
    pub(crate) post: f32,
    /// After PAN: left to the odd buses, right to the even ones; both to the stereo bus.
    pub(crate) left: f32,
    pub(crate) right: f32,
}

/// A channel and its smoothed gains.
#[derive(Debug)]
pub(crate) struct Channel {
    pub(crate) params: ChannelParams,
    trim: f32,
    current: [f32; 2],
}

impl Default for Channel {
    /// TRIM starts at its target so a recording through DIRECT is exact from the first sample;
    /// the fader fades in, as before.
    fn default() -> Self {
        Self { params: ChannelParams::default(), trim: 1.0, current: [0.0; 2] }
    }
}

impl Channel {
    /// Run one sample of the channel's source through it.
    pub(crate) fn process(&mut self, sample: f32) -> Taps {
        smooth(&mut self.trim, self.params.trim);
        let target = if self.params.mute { [0.0; 2] } else { pan_gains(self.params.fader, self.params.pan) };
        smooth(&mut self.current[0], target[0]);
        smooth(&mut self.current[1], target[1]);
        let direct = sample * self.trim;
        // Constant-power pan keeps left² + right² equal to the fader gain squared.
        let post = direct * self.current[0].hypot(self.current[1]);
        Taps { direct, post, left: direct * self.current[0], right: direct * self.current[1] }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn settled(params: ChannelParams) -> Channel {
        let mut channel = Channel { params, ..Channel::default() };
        for _ in 0..20_000 {
            channel.process(0.0);
        }
        channel
    }

    #[test]
    fn trim_comes_before_the_fader_and_pan_splits_left_and_right() {
        let mut channel = settled(ChannelParams { trim: 2.0, fader: 0.5, pan: -1.0, ..ChannelParams::default() });
        let taps = channel.process(0.25);
        assert!((taps.direct - 0.5).abs() < 1e-4, "DIRECT: after TRIM, not the fader");
        assert!((taps.post - 0.25).abs() < 1e-4, "POST: after the fader");
        assert!((taps.left - 0.25).abs() < 1e-4 && taps.right.abs() < 1e-4, "hard left: all to the odd side");
    }

    #[test]
    fn mute_silences_everything_after_the_fader_only() {
        let mut channel = settled(ChannelParams { mute: true, ..ChannelParams::default() });
        let taps = channel.process(1.0);
        assert!((taps.direct - 1.0).abs() < 1e-6);
        assert!(taps.post.abs() < 1e-4 && taps.left.abs() < 1e-4 && taps.right.abs() < 1e-4);
    }

    #[test]
    fn only_mic_or_line_with_an_existing_input_is_live() {
        let line = ChannelParams { source: Source::Line, input: Some(1), ..ChannelParams::default() };
        assert!(line.is_live(2));
        assert!(!line.is_live(1), "the device has no second input");
        assert!(!ChannelParams { input: None, ..line }.is_live(2));
        assert!(!ChannelParams { source: Source::Remix, ..line }.is_live(2));
    }
```

```rust
    /// Whether what track `track` records contains a live source: DIRECT from a live channel,
    /// or a live channel assigned to its bus. Only then does the take need latency compensation;
    /// a bounce from tape lines up with its source as it is.
    fn feeds_live(&self, track: usize) -> bool {
        let channels = self.input.as_ref().map_or(0, |i| i.channels);
        let own = &self.channels[track].params;
        if own.direct {
            own.is_live(channels)
        } else {
            self.channels.iter().any(|c| c.params.assign[track / 2] && c.params.is_live(channels))
        }
    }

    fn punch_in(&mut self, track: usize) {
        if self.recording[track] || !self.armed[track] {
            return;
        }
        // Decided once per take: changing the routing during the take does not move it.
        let compensation = if self.feeds_live(track) { self.compensation } else { 0.0 };
        if self.rec_out.tracks[track].push(RecItem::Start(self.position - compensation)).is_err() {
            self.shared.overflow.store(true, Ordering::Relaxed);
            return;
        }
        self.recording[track] = true;
        self.rec_mode = RecMode::Recording;
    }
```

- [ ] **Step 6: `render` herschrijven**

Vervang de hele `pub fn render` (inclusief doc-commentaar) door:

```rust
    /// Fill an interleaved output buffer with `channels` channels. Channel 0 and 1 get the stereo
    /// master, further channels silence; a mono device gets the average of left and right.
    /// Winding is silent, as on the 388.
    ///
    /// Per frame, as the 388's block diagram: read the tape, run the channels, sum the PGM buses
    /// (once as recorded, once as heard), feed the tracks, mix the MONITOR section and the
    /// stereo bus.
    #[allow(clippy::needless_range_loop)] // the channels index many parallel arrays
    pub fn render(&mut self, out: &mut [f32], channels: usize) {
        self.apply_commands();
        self.update_compensation();
        let channels = channels.max(1);
        let input_channels = self.input.as_ref().map_or(0, |i| i.channels);
        let mut track_peaks = [0.0f32; TRACKS];
        let mut track_peaks_post = [0.0f32; TRACKS];
        let mut trk_peaks = [0.0f32; TRACKS];
        let mut master_peaks = [0.0f32; 2];

        for frame in out.chunks_mut(channels) {
            let playing = self.state == State::Playing;
            if let Some(input) = &mut self.input {
                input.next_frame();
            }
            // Every sum exists twice: what is recorded, and what is heard. They differ only when
            // the interface does the monitoring: then live sources are recorded but not heard.
            let mut tape = [0.0f32; TRACKS];
            let mut bus_recorded = [0.0f32; TRACKS];
            let mut bus_heard = [0.0f32; TRACKS];
            let mut direct = [(0.0f32, 0.0f32); TRACKS];
            let mut stereo = [0.0f32; 2];
            let mut any_lr = false;
            for index in 0..TRACKS {
                if playing && let Some(audio) = &self.tape[index] {
                    tape[index] = sample_at(&audio.samples, self.position);
                }
                let params = self.channels[index].params;
                let live = params.is_live(input_channels);
                let sample = match params.source {
                    // A track that hears its input is not played back into its own channel, so
                    // bus, track and channel never form a loop.
                    Source::Remix if hears_input(self.armed[index], self.insert, self.recording[index]) => 0.0,
                    Source::Remix => tape[index],
                    Source::Mic | Source::Line => match (&self.input, params.input) {
                        (Some(input), Some(channel)) if live => input.sample(usize::from(channel)),
                        _ => 0.0,
                    },
                };
                let taps = self.channels[index].process(sample);
                track_peaks[index] = track_peaks[index].max(taps.direct.abs());
                track_peaks_post[index] = track_peaks_post[index].max(taps.post.abs());
                let audible = if live && !self.monitoring { 0.0 } else { 1.0 };
                direct[index] = (taps.direct, taps.direct * audible);
                for (pair, on) in params.assign.into_iter().enumerate() {
                    if on {
                        bus_recorded[2 * pair] += taps.left;
                        bus_recorded[2 * pair + 1] += taps.right;
                        bus_heard[2 * pair] += taps.left * audible;
                        bus_heard[2 * pair + 1] += taps.right * audible;
                    }
                }
                if params.lr {
                    any_lr = true;
                    stereo[0] += taps.left * audible;
                    stereo[1] += taps.right * audible;
                }
            }
            smooth(&mut self.bus_gains[0], self.mixer.odd);
            smooth(&mut self.bus_gains[1], self.mixer.even);
            // Track N records PGM bus N (or the channel's DIRECT tap); its MONITOR hears the input,
            // the tape or nothing, as the 388's switching logic decides.
            let mut monitor = [0.0f32; 2];
            for index in 0..TRACKS {
                let (recorded, heard) = if self.channels[index].params.direct {
                    direct[index]
                } else {
                    // Buses 1, 3, 5 and 7 go through the ODD master, 2, 4, 6 and 8 through EVEN.
                    let master = self.bus_gains[index % 2];
                    (bus_recorded[index] * master, bus_heard[index] * master)
                };
                if self.recording[index] && self.rec_out.tracks[index].push(RecItem::Sample(recorded)).is_err() {
                    self.shared.overflow.store(true, Ordering::Relaxed);
                }
                let value = match monitor_source(self.mixer.mon, self.armed[index], self.insert, self.recording[index], playing) {
                    MonitorSource::Input => heard,
                    MonitorSource::Tape => tape[index],
                    MonitorSource::Silent => 0.0,
                };
                trk_peaks[index] = trk_peaks[index].max(value.abs());
                let gains = self.monitors[index].gains();
                monitor[0] += value * gains[0];
                monitor[1] += value * gains[1];
            }
            // The STEREO MASTER takes the L-R channels, or the MONITOR mix when none is on L-R.
            if !any_lr {
                stereo[0] += monitor[0];
                stereo[1] += monitor[1];
            }
            // The click follows the tape's bar grid; during the count-in the grid runs ahead of the
            // waiting tape, so the beats carry on into the recording without a jump.
            let (click_at, sounding) = match self.state {
                State::Playing => (self.position, self.click.enabled && (self.click.always || self.rec_mode != RecMode::Off)),
                State::CountIn { left } => (self.position - left, true),
                _ => (0.0, false),
            };
            self.click.follow(click_at, sounding);
            self.advance();
            smooth(&mut self.master_current, self.master);
            let (mut left, mut right) = (stereo[0] * self.master_current, stereo[1] * self.master_current);
            if self.mixer.mono {
                let mono = 0.5 * (left + right);
                (left, right) = (mono, mono);
            }
            master_peaks[0] = master_peaks[0].max(left.abs());
            master_peaks[1] = master_peaks[1].max(right.abs());
            // The click goes to the output only: not into the master meters or a recording.
            let click = self.click.next();
            let (left, right) = (left + click, right + click);
            match frame {
                [mono] => *mono = 0.5 * (left + right),
                [l, r, rest @ ..] => {
                    *l = left;
                    *r = right;
                    rest.fill(0.0);
                }
                [] => {}
            }
        }

        self.shared.position.store(self.position.to_bits(), Ordering::Relaxed);
        self.shared.zero.store(self.zero.to_bits(), Ordering::Relaxed);
        self.shared.cue.store(self.cue.map_or(NO_CUE, f64::to_bits), Ordering::Relaxed);
        self.shared.motion.store(motion_code(self.motion()), Ordering::Relaxed);
        self.shared.rec_mode.store(rec_mode_code(self.rec_mode), Ordering::Relaxed);
        let mask = self.recording.iter().enumerate().fold(0u8, |m, (i, r)| if *r { m | (1 << i) } else { m });
        self.shared.recording.store(mask, Ordering::Relaxed);
        self.shared.compensation.store((self.compensation / f64::from(self.sample_rate)).to_bits(), Ordering::Relaxed);
        let count_in = if let State::CountIn { left } = self.state { left / f64::from(self.sample_rate) } else { 0.0 };
        self.shared.count_in.store(count_in.to_bits(), Ordering::Relaxed);
        for (cell, peak) in self.shared.track_peaks.iter().zip(track_peaks) {
            store_peak(cell, peak);
        }
        for (cell, peak) in self.shared.track_peaks_post.iter().zip(track_peaks_post) {
            store_peak(cell, peak);
        }
        for (cell, peak) in self.shared.trk_peaks.iter().zip(trk_peaks) {
            store_peak(cell, peak);
        }
        for (cell, peak) in self.shared.master_peaks.iter().zip(master_peaks) {
            store_peak(cell, peak);
        }
    }
```

```rust
/// TRIM range per source: more gain on MIC, as on the 388's microphone preamp.
fn trim_range(source: ChannelSource) -> std::ops::RangeInclusive<f32> {
    match source {
        ChannelSource::Mic => 0.0..=40.0,
        ChannelSource::Line | ChannelSource::Rmx => -10.0..=20.0,
    }
}

fn channel_params(track: &TrackState) -> ChannelParams {
    ChannelParams {
        source: match track.channel.source {
            ChannelSource::Mic => Source::Mic,
            ChannelSource::Line => Source::Line,
            ChannelSource::Rmx => Source::Remix,
        },
        // The engine counts input channels from 0.
        input: track.input.and_then(|c| c.checked_sub(1)),
        trim: db_to_gain(track.channel.trim_db),
        fader: db_to_gain(track.fader_db),
        pan: track.pan,
        mute: track.mute,
        assign: track.channel.assign,
        lr: track.channel.lr,
        direct: track.channel.direct,
    }
}

fn monitor_params(monitor: &MonitorEntry) -> MonitorParams {
    MonitorParams { gain: db_to_gain(monitor.gain_db), pan: monitor.pan, mute: monitor.mute }
}

fn mixer_params(mixer: &MixerEntry) -> MixerParams {
    let mon = match mixer.mon {
        MonSwitch::Pgm => MonitorMode::Pgm,
        MonSwitch::Cue => MonitorMode::Cue,
    };
    MixerParams { odd: db_to_gain(mixer.odd_db), even: db_to_gain(mixer.even_db), mon, mono: mixer.mono }
}
```

```rust
/// A rotary knob: drag up or down to turn it (Shift for fine), double-click for `default`.
/// `enabled` false draws it dimmed (the pitch control in FIX).
fn knob(
    ui: &mut Ui,
    value: &mut f32,
    range: std::ops::RangeInclusive<f32>,
    default: f32,
    colour: Color32,
    size: f32,
    enabled: bool,
) -> egui::Response {
    let (rect, mut response) = ui.allocate_exact_size(vec2(size, size), Sense::click_and_drag());
    let (min, max) = (*range.start(), *range.end());
    if enabled && response.dragged() {
        let fine = if ui.input(|i| i.modifiers.shift) { 0.2 } else { 1.0 };
        let delta = -response.drag_delta().y / 150.0 * (max - min) * fine;
        *value = (*value + delta).clamp(min, max);
        response.mark_changed();
    }
    if enabled && response.double_clicked() {
        *value = default;
    }
    let painter = ui.painter();
    let centre = rect.center();
    let radius = size / 2.0 - 2.0;
    let cap = if enabled { colour } else { Color32::from_rgb(0x8a, 0x86, 0x7c) };
    painter.circle_filled(centre, radius + 2.0, Color32::from_rgb(0x1b, 0x1b, 0x1c));
    painter.circle_filled(centre, radius, cap);
    // From -135 to +135 degrees, 0 at the top.
    let fraction = (*value - min) / (max - min);
    let angle = (-135.0 + 270.0 * fraction).to_radians();
    let direction = vec2(angle.sin(), -angle.cos());
    painter.line_segment(
        [centre + direction * (radius * 0.3), centre + direction * (radius * 0.9)],
        Stroke::new(2.5, Color32::from_rgb(0x2a, 0x26, 0x22)),
    );
    response
}
```

```rust
/// A small horizontal peak meter on the same scale and colours as [`meter`].
fn level_bar(ui: &mut Ui, db: f32) {
    let (rect, _) = ui.allocate_exact_size(vec2(36.0, 6.0), Sense::hover());

    painter.rect_filled(rect, 1.0, Color32::BLACK);

        palette::GREEN
    };
```

Onder `## [Unreleased]` (de bestaande CI-regel blijft daar), een nieuwe sectie `## [0.8.0] - 2026-10-28`:

```markdown
### Toegevoegd
- Mixer naar het blokschema van de 388. Per kanaal INPUT MIC, LINE of RMX (de band), TRIM, een PAN-knop en ASSIGN 1-2, 3-4, 5-6 en 7-8 naar de PGM-bussen: PAN links gaat naar de oneven bus, rechts naar de even bus.
- PGM MASTERS ODD en EVEN. Spoor N neemt PGM-bus N op, dus ping-pongen van sporen naar een vrij spoor kan nu.
- DIRECT per spoor: het spoor neemt zijn eigen kanaal op (na TRIM, voor de fader), zonder bus.
- MONITOR-sectie met per spoor GAIN, PAN, mute en een TRK/PGM-meter. Wat je hoort volgt de omschakeltabel van de 388: MON op CUE geeft band of ingang volgens REC FUNCTION en INSERT, op PGM altijd de bussen.
- L-R per kanaal naar de stereobus, voor de remix. Staat er een kanaal op L-R, dan hoor je alleen de L-R-kanalen.
- MONO-schakelaar bij de STEREO MASTER.
- Opnemen kan nu ook zonder ingangsapparaat, van band naar band. Zo'n bounce ligt sample-precies onder de bron; alleen opnames van een ingang krijgen latencycompensatie.

### Gewijzigd
- `project.json` heeft formaatversie 2. Een project van een eerdere versie gaat open zoals het klonk: de oude spoorfader, pan en mute staan nu in de MONITOR-sectie, de kanaalfader staat op 0 dB. Een spoor met een ingang (in oude projecten is dat elk spoor) wordt een LINE-kanaal met DIRECT, zodat opnemen werkt als voorheen. Daardoor tonen de kanaalmeters na de omzetting de ingang in plaats van de band; de TRK/PGM-meters in de MONITOR-sectie tonen wat je hoort. Zet een kanaal op RMX om de band weer op de kanaalmeter te zien.
- Een spoor op REC FUNCTION met INSERT aan is stil zolang de band stilstaat, zoals op de 388. Zet MON op PGM om dan de ingang te horen.
- REC weigert niet meer zonder ingang, maar waarschuwt als een klaargezet spoor niets binnenkrijgt.
- De ingang kies je nu per kanaal (voor MIC en LINE). Het venster is hoger en scrolt ook verticaal.
```

```markdown
## Mixer

Naar het blokschema en Features and Controls in de handleiding van de 388:

- Kanaal: bron (MIC, LINE of RMX), TRIM, fader, PAN, ASSIGN 1-2/3-4/5-6/7-8 en L-R. RMX is de weergave van het spoor met hetzelfde nummer; zolang dat spoor zijn ingang hoort (REC FUNCTION aan en opnemend of zonder INSERT) is RMX stil, zodat er geen lus ontstaat.
- PAN verdeelt over de oneven (links) en even (rechts) bus van een ASSIGN-paar. De PGM MASTERS ODD en EVEN regelen bussen 1, 3, 5, 7 en 2, 4, 6, 8.
- Spoor N neemt PGM-bus N op, of met DIRECT kanaal N na TRIM (en straks EQ), voor de fader. DIRECT is een afwijking van de 388: een snelle opnameroute naast de bussen.
- MONITOR: per spoor GAIN, PAN en mute. Met MON op PGM hoort de monitor de bus (of het DIRECT-signaal); op CUE volgt hij de tabel uit de handleiding: de ingang zolang het spoor zijn ingang hoort, anders de band als die speelt, anders niets.
- De STEREO MASTER krijgt de L-R-kanalen, of de MONITOR-mix als geen kanaal op L-R staat. Daarna MONO.
- Meeluisteren via de interface: live bronnen worden wel opgenomen maar nergens gehoord; de engine houdt per bus een opname- en een luistersom bij.
- Een nieuw project staat in de opnamestand: alle kanalen RMX zonder ASSIGN, je hoort de sporen via de MONITOR.
- Latencycompensatie geldt alleen als de opname een live bron bevat (bepaald bij de punch-in).
```

- Bekende beperkingen: vervang "Geen opname, EQ, sends, effecten of tape-model." door "Nog geen EQ, sends, effecten of tape-model (zie de roadmap)." en voeg toe: "Een bus die een live ingang en de band tegelijk opneemt, wordt gecompenseerd voor de ingang; het banddeel komt dan de apparaatlatency te vroeg op het nieuwe spoor."
